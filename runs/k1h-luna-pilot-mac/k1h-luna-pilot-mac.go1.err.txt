    raw=p.read_bytes()
    crlf=raw.count(b"\r\n")
    lines=raw.decode("utf-8").splitlines()
    n_lines=len(lines)

    err_counter=collections.Counter()
    ids=set()
    for ln in lines:

        try:
            r=json.loads(ln)
        except Exception as e:
            err_counter[f"json_error:{e}"]+=1
            continue
